\SlideHead{6. Late Fusion \& Cổng Lọc Cờ Đỏ Cấp Cứu}{Cơ chế hòa trộn động đa phương thức và bộ lọc khẩn cấp thời gian thực $\le 50$ms}{SLIDE 08}

\begin{minipage}[t]{0.49\textwidth}
\Lone{Cơ Chế Hòa Trộn Xác Suất Late Fusion}
\Ltwo{Xác suất tổng hợp cuối cùng được tính theo công thức hàm trọng số thích ứng:}
\begin{equation*}
  P_{\text{final}}(c) = \alpha \cdot P_{\text{tabular}}(c) + (1 - \alpha) \cdot P_{\text{text}}(c)
\end{equation*}
\Ltwo{\textbf{Trọng số động $\alpha$}:
  \begin{itemize}
    \setlength{\itemsep}{0pt}
    \item $\alpha = 0.65$: Khi có kết quả xét nghiệm máu (ưu tiên chỉ số định lượng khách quan).
    \item $\alpha = 0.00$: Khi chỉ có lời khai triệu chứng từ bệnh nhân.
  \end{itemize}
}
\Ltwo{\textbf{Penalization Phủ định}: Các bệnh xuất hiện trong tập triệu chứng phủ định bị phạt trừ xác suất tự động.}
\end{minipage}%
\hfill%
\begin{minipage}[t]{0.48\textwidth}
\Lone{Cổng Lọc Cờ Đỏ Cấp Cứu (Red Flag Rapid Filter)}
\Ltwo{Chạy song song ở tầng đầu vào với \textbf{độ trễ siêu tốc $\le 50\text{ ms}$}.}
\Ltwo{\textbf{FAST Protocol (Đột quỵ não)}: Méo miệng, lệch mặt, yếu liệt nửa người, nói ngọng $\to$ kích hoạt cảnh báo đột quỵ khẩn.}
\Ltwo{\textbf{STEMI/NSTEMI (Nhồi máu cơ tim)}: Đau ngực đè ép dữ dội lan tay trái/hàm, vã mồ hôi, khó thở hoặc Troponin tăng.}
\Ltwo{\textbf{Dengue Shock}: Xuất huyết chân răng/chảy máu cam + sốt cao kéo dài + tiểu cầu hạ cấp tính ($< 50 \text{ G/L}$).}
\Ltwo{\textbf{Hành động can thiệp}: Hiển thị khung báo động đỏ, nút gọi cấp cứu 115 một chạm và cẩm nang sơ cứu tại chỗ.}

\SlideHead{7. Động Cơ Shannon Entropy \& Làm Rõ}{Thuật toán đo lường độ bất định thông tin và điều phối câu hỏi gạn lọc}{SLIDE 09}

\begin{minipage}[t]{0.49\textwidth}
\Lone{Định Lượng Độ Bất Định Lâm Sàng}
\Ltwo{Khi bệnh nhân cung cấp thông tin chung chung (\textit{"thấy mệt mỏi, khó chịu"}), xác suất dự đoán bị phân tán đều trên nhiều bệnh.}
\Ltwo{Hệ thống tính toán \textbf{Độ hỗn loạn thông tin Shannon (Shannon Entropy)} trên Top-$K$ phân phối xác suất dự đoán:}
\vspace{0.02cm}
{\centering\color{samblue}$H(P) = -\sum_{i=1}^{K} p_i \log_2(p_i)$\par}
\vspace{0.02cm}
\Ltwo{\textbf{Điều kiện kích hoạt Clarification Loop}:
  \begin{itemize}
    \setlength{\itemsep}{0pt}
    \item $H(P) > \theta = 1.8$ nats (độ hỗn loạn vượt ngưỡng an toàn).
    \item Hoặc khoảng cách xác suất Top-1 và Top-2: $\Delta p = p_{(1)} - p_{(2)} < 0.15$.
  \end{itemize}
}
\end{minipage}%
\hfill%
\begin{minipage}[t]{0.48\textwidth}
\Lone{Cơ Chế Sinh Câu Hỏi Phân Biệt}
\Ltwo{Tự động tra cứu ngân hàng câu hỏi lâm sàng theo cặp bệnh cần phân biệt (Differential Diagnosis).}
\Ltwo{Lựa chọn 2--3 câu hỏi có \textbf{Information Gain} cao nhất về triệu chứng khu biệt quan trọng.}
\Ltwo{Sau phản hồi, cập nhật xác suất Bayes, thu hẹp không gian chẩn đoán nhanh chóng.}

\vspace{0.05cm}
